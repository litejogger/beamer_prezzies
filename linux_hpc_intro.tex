\section{Knowing When to Use HPC or VCL}

% ---- Slide: HPC vs VCL table ----
\begin{frame}{HPC Cluster vs.\ Large Linux Machine (VCL)}
  \begin{table}
    \centering
    \small
    \begin{tabular}{l|l|l}
      \toprule
      & \textbf{HPC Cluster (Henry2)} & \textbf{VCL (1\,TB RAM, 176 cores)} \\
      \midrule
      Architecture    & Many nodes, each modest           & One machine, massive shared memory \\
      \midrule
      Access          & Submit jobs via LSF scheduler     & Interactive login, no queue \\
      \midrule
      RAM per job     & Typically 4--64\,GB per node      & Up to 1\,TB in a single process \\
      \midrule
      Parallelism     & Across nodes (distributed)        & Within one machine (shared memory) \\
      \midrule
      Best for        & Many independent runs             & One large, memory-intensive solve \\
      \bottomrule
    \end{tabular}
  \end{table}
\end{frame}

% ---- Slide: When to use which ----
\begin{frame}{My Own Rules of Thumb}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Use HPC when\ldots}
        \begin{itemize}
          \item Running a \textbf{parameter sweep} --- 100s-1000s of LP/MIP runs at once
          \item Jobs are \textbf{independent} of each other
          \item Larger jobs that require more RAM than your local machine or VCL
          \item You have the time to get familiar with the cluster and its job scheduler
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Use VCL when\ldots}
        \begin{itemize}
          \item Solving a \textbf{large MILP} --- branch-and-bound trees grow into hundreds of GB
          \item Problem cannot be decomposed into independent tasks
          \item You need extra RAM but low setup time and no queue
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\section{The Basics of Getting Started: The Shell and 
The Linux Kernel}

% ---- Slide 2: What Is Linux? ----
\begin{frame}{What Is Linux?}
  The \textbf{kernel} is the core of the operating system ---
  the layer between hardware and your programs.
  \vspace{0.4em}
  \begin{itemize}
    \item Manages CPU scheduling, memory, storage, networking, and I/O
    \item Every process running on the cluster goes through it
    \item You interact with it through a \textbf{shell} (bash)
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{The software stack}
    \textbf{Hardware} $\to$ \textbf{Linux Kernel} $\to$ \textbf{Bash Shell} $\to$ \textbf{Your Code}
  \end{block}
\end{frame}

% ---- Slide 6: Why Linux for HPC ----
\begin{frame}{Why Linux for HPC}
  \begin{itemize}
    \item \textbf{Free and open source} --- every HPC cluster in the world runs Linux
    \item \textbf{Stable under heavy load} --- built for long-running, multi-user environments
    \item \textbf{Shell scripting is built in} --- automate repetitive tasks with a text file
    \item \textbf{Environment management} --- \texttt{conda activate}, \texttt{module load} work natively
    \item \textbf{Remote access} --- SSH from any machine, no GUI required
  \end{itemize}
  \vspace{0.5em}
  \begin{alertblock}{Windows paths do not work here}
    \texttt{C:\textbackslash{}Users\textbackslash{}calisy\textbackslash{}...} is a Windows path.
    On Linux everything uses forward slashes and starts from \texttt{/}.\\[0.3em]
    \texttt{\$HOME}, \texttt{\$GROUP}, and \texttt{\$USER} are \textbf{environment variables} ---
    shell variables defined in your \texttt{.bashrc} (the config file that runs every time you log in).\\[0.2em]
    \texttt{\$HOME} = your home directory \quad \texttt{/share/\$GROUP/\$USER/} = scratch storage
  \end{alertblock}
\end{frame}% ---- Slide: File System Hierarchy ----
\begin{frame}{The Linux File System Hierarchy}
  \begin{itemize}
    \item \texttt{/} --- \textbf{Root}: The top-level directory (like \texttt{C:\textbackslash} but for everything)
    \item \texttt{/home/\$USER} --- \textbf{Home}: Your personal space for code and small files
    \item \texttt{/share/\$GROUP} --- \textbf{Scratch}: High-performance storage for large datasets
    \item \texttt{/usr/local/usrapps} --- \textbf{Apps}: Where shared software and environments live
    \item \texttt{..} --- \textbf{Parent}: Move up one directory level
    \item \texttt{.} --- \textbf{Current}: The directory you are currently in
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Pro Tip: Tab Completion}
    Start typing a file or directory name and hit \textbf{Tab}. 
    Linux will auto-complete it for you. Hit \textbf{Tab} twice to see all possibilities.
  \end{block}
\end{frame}% ---- Slide: Absolute vs Relative Paths ----
\begin{frame}{Absolute vs Relative Paths}
  \begin{itemize}
    \item \textbf{Absolute path} starts from root: \texttt{/home/\$USER/project/script.py}
    \item \textbf{Relative path} starts from current directory: \texttt{./script.py} or \texttt{../data/input.txt}
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Best Practice}
    Use absolute paths in scripts to avoid confusion.
    For interactive work, relative paths can be more convenient.
  \end{block}
\end{frame}



% ---- Slide 3: Key Terms ----
\begin{frame}{Key Terms}
  \begin{block}{On Henry2}
    When you SSH into the cluster, you land in a bash shell.
    Commands you type go to the kernel, which runs them on the hardware.
  \end{block}
  \vspace{0.5em}
  \begin{itemize}
    \item \textbf{SSH (Secure Shell)} = the protocol to securely connect to the cluster from your local computer
    \item \textbf{Linux} = the kernel + surrounding tools
    \item \textbf{Bash} = the shell you type commands into
    \item \textbf{LSF} = the job scheduler that sits on top
  \end{itemize}
\end{frame}

% ---- Slide 3: Why Learn the Shell? ----
\begin{frame}{Why Learn the Shell Instead of a UI?}
  Some clusters offer a web portal or file browser --- but working directly in the shell is worth the learning curve:
  \vspace{-0.1em}
  \begin{itemize}
    \item \textbf{Speed} --- scripts run instantly; UIs require clicking through multiple pages
    \item \textbf{Automation} --- a UI requires a human every time; a bash script runs while you sleep
    \item \textbf{Reproducibility} --- a script is a record of exactly what you ran; clicks are not
    \item \textbf{Power} --- chaining commands (\texttt{|}, \texttt{\&\&}, loops) lets you do in one line what takes dozens of UI steps
  \end{itemize}
  \vspace{-0.1em}
  \begin{block}{The payoff}
    The shell has a steep learning curve initially. After that, you will not want to go back.
  \end{block}
\end{frame}

\section{Bash Scripting}

% ---- Slide 5: Anatomy of a Bash Script ----
\begin{frame}[fragile]{Anatomy of a Bash Script}
  \begin{columns}[T]
    \column{0.42\textwidth}
      \begin{block}{Key rules}
        \begin{enumerate}
          \item \texttt{\#!/bin/bash} must be line~1
          \item No spaces around \texttt{=} in assignments
          \item Use \texttt{\$VAR} or \texttt{\$\{VAR\}} to expand a variable
          \item Make executable before first run:\\
            \texttt{chmod +x script.sh}
        \end{enumerate}
      \end{block}
      \vspace{0.1em}
      \begin{block}{Three ways to run}
        \begin{itemize}
          \item \texttt{./script.sh} \quad subprocess
          \item \texttt{bash script.sh} \quad same
          \item \texttt{. ./script.sh} \quad \textit{source} --- runs in current shell and imports its variables
        \end{itemize}
      \end{block}

    \column{0.54\textwidth}
      \begin{lstlisting}[language=bash]
#!/bin/bash
# this is a comment

# source a config file (imports its variables)
. ./experiment_grid_params.sh

# variable assignment (no spaces around =)
MY_VAR="hello world"

# print a variable
echo $MY_VAR

# loop over an array
for val in "${my_array[@]}"; do
    echo "Running: $val"
    python run_model.py $val
done
      \end{lstlisting}
  \end{columns}
\end{frame}

\section{HPC with LSF}

% ---- Slide 6: LSF Submission Script ----
\begin{frame}[fragile]{One-Line LSF Job Submission with \texttt{bsub}}
  \begin{lstlisting}[language=bash, basicstyle=\ttfamily\scriptsize]
bsub -W 30 -n 1 -o "out.%J" -e "err.%J" "python v13_calibration_wrapper.py"
  \end{lstlisting}
  % \vspace{0.2em}
  \begin{table}
    \centering
    \small
    \begin{tabular}{lll}
      \toprule
      \textbf{Flag} & \textbf{Meaning} & \textbf{Typical value} \\
      \midrule
      \texttt{-W 30}      & Wall time limit (minutes)       & \texttt{30} for model run; \texttt{5} for data setup \\
      \texttt{-n 1}       & Number of CPU cores             & \texttt{1} (model is single-threaded) \\
      \texttt{-o out.\%J} & Write stdout to this file       & \texttt{\%J} expands to the LSF job ID \\
      \texttt{-e err.\%J} & Write stderr to this file       & Read this first when a job fails \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{1em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \vspace{0.2em}
      \textbf{How to Monitor your jobs:}
      \begin{itemize}
        \item \texttt{bjobs -r} \quad\quad running jobs
        \item \texttt{bjobs -d} \quad\quad recently finished
        \item \texttt{bhist -l \$JOBID} \quad full history
      \end{itemize}
    \column{0.48\textwidth}
      \begin{block}{If a job fails}
        Check \texttt{err.\$JOBID} first.
        Common causes: wall time exceeded (\texttt{-W} too short),
        wrong conda env, or a missing input file.
      \end{block}
  \end{columns}
\end{frame}

% ---- Slide: Putting It Together ----
\begin{frame}[fragile]{Putting It Together: A Single Job Submission}
  \begin{lstlisting}[language=bash]
#!/bin/bash

# activate the group conda environment
conda activate /usr/local/usrapps/infews/group_env_111425

# submit one model run to the cluster
bsub -W 30 -n 1 -o "out.%J" -e "err.%J" "python v13_calibration_wrapper.py"

echo "Job submitted"
  \end{lstlisting}
  \vspace{0.3em}
  \begin{block}{What this does}
    \begin{enumerate}\small
      \item Activates the conda environment so Python and all packages are available
      \item Hands the job to LSF --- the scheduler picks a free compute node and runs it
      \item Stdout goes to \texttt{out.\$JOBID}, stderr to \texttt{err.\$JOBID} --- check these if it fails
    \end{enumerate}
  \end{block}
\end{frame}
